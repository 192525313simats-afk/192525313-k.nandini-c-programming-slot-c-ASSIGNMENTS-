#include <stdio.h>
int main() {
    int principal, rate, years;
    scanf("%d %d %d", &principal, &rate, &years);

    float interest = (principal * rate * years) / 100.0;
    float amount = principal + interest;
    float discount = interest * 0.02;
    float finalSettlement = amount - discount;

    printf("%.2f\n%.2f\n%.2f\n%.2f\n", interest, amount, discount, finalSettlement);
    return 0;
}
